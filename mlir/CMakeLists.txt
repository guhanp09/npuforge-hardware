cmake_minimum_required(VERSION 3.20)
project(NPUForgeMLIR LANGUAGES C CXX)
if(POLICY CMP0156)
  cmake_policy(SET CMP0156 NEW)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(MLIR REQUIRED CONFIG)
if(NOT LLVM_VERSION_MAJOR EQUAL 19)
  message(FATAL_ERROR "NPUForge MLIR currently supports LLVM/MLIR 19.x")
endif()
list(APPEND CMAKE_MODULE_PATH "${MLIR_CMAKE_DIR}" "${LLVM_CMAKE_DIR}")
include(TableGen)
include(AddLLVM)
include(AddMLIR)
include_directories(SYSTEM ${LLVM_INCLUDE_DIRS} ${MLIR_INCLUDE_DIRS})
include_directories(include ${CMAKE_CURRENT_BINARY_DIR})
add_definitions(${LLVM_DEFINITIONS})
set(LLVM_TARGET_DEFINITIONS include/NPUForge.td)
mlir_tablegen(NPUForgeOps.h.inc -gen-op-decls)
mlir_tablegen(NPUForgeOps.cpp.inc -gen-op-defs)
mlir_tablegen(NPUForgeDialect.h.inc -gen-dialect-decls)
mlir_tablegen(NPUForgeDialect.cpp.inc -gen-dialect-defs)
add_public_tablegen_target(NPUForgeIncGen)
add_executable(npuforge-opt npuforge-opt.cpp lib/Dialect.cpp lib/Passes.cpp
               lib/Lowering.cpp lib/Export.cpp)
add_dependencies(npuforge-opt NPUForgeIncGen)
target_compile_options(npuforge-opt PRIVATE -Wall -Wextra -Wpedantic -Wno-unused-parameter)
target_link_libraries(npuforge-opt PRIVATE MLIROptLib MLIRTransforms
  MLIRFuncDialect MLIRLinalgDialect)
install(TARGETS npuforge-opt RUNTIME DESTINATION bin)
